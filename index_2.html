/** * PLS HQ — WhatsApp Command Centre — Cloud Functions backend (Phase 1) * * Two functions: * - ingestWhatsAppEvent (HTTP) called by the existing Wix webhook, mirrors * incoming messages + status updates into Firestore * - sendWhatsAppReply (Callable) called by PLS HQ (signed-in agents only), * enforces the 24h window, then calls the * EXISTING Wix sendReply/sendTemplate endpoints — * this function never talks to Meta directly and * never touches the Meta token. * * Nothing about the existing Wix→Meta sending path changes. This is a relay + mirror. */ const { onRequest, onCall, HttpsError } = require("firebase-functions/v2/https"); const { defineSecret } = require("firebase-functions/params"); const logger = require("firebase-functions/logger"); const admin = require("firebase-admin"); admin.initializeApp(); const db = admin.firestore(); // Shared secret, used BOTH ways: Wix→here (ingest) and here→Wix (send). // Set via: firebase functions:secrets:set BRIDGE_SECRET const BRIDGE_SECRET = defineSecret("BRIDGE_SECRET"); // Your Wix site's base URL for the existing, unchanged endpoints. const WIX_BASE = "https://www.packlifestore.co.za/_functions"; const WINDOW_MS = 24 * 60 * 60 * 1000; function cleanPhone(p) { if (!p) return null; let c = String(p).replace(/\D/g, ""); if (c.startsWith("0")) c = "27" + c.substring(1); return c.length < 10 ? null : c; } // ───────────────────────────────────────────────────────────────────────── // ingestWhatsAppEvent — called by Wix's post_whatsappWebhook for every // incoming message and every status update. Purely additive/mirroring; // if this fails, Wix's own existing storage (IncomingMessages) is unaffected. // ───────────────────────────────────────────────────────────────────────── exports.ingestWhatsAppEvent = onRequest( { secrets: [BRIDGE_SECRET], cors: false, region: "us-central1" }, async (req, res) => { try { if (req.method !== "POST") { res.status(405).json({ error: "POST only" }); return; } if (req.get("X-Bridge-Secret") !== BRIDGE_SECRET.value()) { res.status(401).json({ error: "Unauthorized" }); return; } const body = req.body || {}; if (body.type === "message") { const phone = cleanPhone(body.from); if (!phone) { res.status(400).json({ error: "Missing/invalid from" }); return; } const messageId = body.messageId || `local_${Date.now()}_${Math.random().toString(36).slice(2)}`; const now = admin.firestore.Timestamp.now(); const convRef = db.collection("whatsapp_conversations").doc(phone); const msgRef = convRef.collection("messages").doc(messageId); await db.runTransaction(async (tx) => { const existing = await tx.get(msgRef); if (existing.exists) { logger.info("Duplicate inbound message ignored", messageId); return; } const convSnap = await tx.get(convRef); const prevUnread = convSnap.exists ? (convSnap.data().unreadCount || 0) : 0; tx.set(msgRef, { direction: "in", type: body.messageType || "text", text: body.text || "[unsupported message type]", status: "received", source: "customer", createdAt: now, mediaUrl: body.mediaUrl || null, mediaFileRef: body.mediaFileRef || null, mediaType: body.mediaType || null, }); tx.set(convRef, { phone, customerName: body.customerName || phone, lastMessageText: body.text || "[unsupported message type]", lastMessageAt: now, lastMessageDirection: "in", unreadCount: prevUnread + 1, windowExpiresAt: admin.firestore.Timestamp.fromMillis(Date.now() + WINDOW_MS), status: "open", updatedAt: now, }, { merge: true }); }); res.json({ success: true }); return; } if (body.type === "status") { const phone = cleanPhone(body.recipientPhone); const messageId = body.messageId; if (!phone || !messageId) { res.status(400).json({ error: "Missing recipientPhone/messageId" }); return; } const msgRef = db.collection("whatsapp_conversations").doc(phone).collection("messages").doc(messageId); const snap = await msgRef.get(); if (!snap.exists) { // Status arrived before we'd logged the outbound message (rare race) — store it anyway await msgRef.set({ status: body.status, statusUpdatedAt: admin.firestore.Timestamp.now() }, { merge: true }); } else { await msgRef.update({ status: body.status, statusUpdatedAt: admin.firestore.Timestamp.now() }); } res.json({ success: true }); return; } res.status(400).json({ error: "Unknown event type" }); } catch (err) { logger.error("ingestWhatsAppEvent error", err); res.status(500).json({ error: err.message }); } } ); // ───────────────────────────────────────────────────────────────────────── // sendWhatsAppReply — called from PLS HQ by a signed-in agent. // Enforces the 24h window, then calls the EXISTING Wix endpoints unchanged. // ───────────────────────────────────────────────────────────────────────── exports.sendWhatsAppReply = onCall( { secrets: [BRIDGE_SECRET], region: "us-central1" }, async (request) => { if (!request.auth) { throw new HttpsError("unauthenticated", "Sign in required."); } const agentEmail = request.auth.token.email || request.auth.uid; const { phone: rawPhone, mode, text, templateName, language, variables, customerName, renderedText } = request.data || {}; const phone = cleanPhone(rawPhone); if (!phone) throw new HttpsError("invalid-argument", "Missing or invalid phone number."); const convRef = db.collection("whatsapp_conversations").doc(phone); const convSnap = await convRef.get(); const conv = convSnap.exists ? convSnap.data() : null; if (mode === "text") { if (!text || !text.trim()) throw new HttpsError("invalid-argument", "Message text is required."); const windowExpiresAt = conv && conv.windowExpiresAt ? conv.windowExpiresAt.toMillis() : 0; if (Date.now() > windowExpiresAt) { throw new HttpsError( "failed-precondition", "24-hour reply window expired. Send an approved template to reopen this conversation." ); } const res = await fetch(`${WIX_BASE}/sendReply`, { method: "POST", headers: { "Content-Type": "application/json", "X-Bridge-Secret": BRIDGE_SECRET.value() }, body: JSON.stringify({ to: phone, message: text.trim(), customerName: customerName || conv?.customerName || phone }), }); const result = await res.json(); if (!res.ok || result.success === false) { throw new HttpsError("internal", "Message failed to send: " + JSON.stringify(result.error || result)); } const now = admin.firestore.Timestamp.now(); const outMsgId = `out_${Date.now()}_${Math.random().toString(36).slice(2)}`; await convRef.collection("messages").doc(outMsgId).set({ direction: "out", type: "text", text: text.trim(), status: "sent", source: "manual", sentBy: agentEmail, createdAt: now, }); await convRef.set({ lastMessageText: text.trim(), lastMessageAt: now, lastMessageDirection: "out", unreadCount: 0, updatedAt: now, }, { merge: true }); return { success: true }; } if (mode === "template") { if (!templateName) throw new HttpsError("invalid-argument", "templateName is required."); const firstName = (customerName || conv?.customerName || "there").split(" ")[0] || "there"; const payload = { to: phone, template: templateName, language: language || "en", customer_name: firstName }; Object.assign(payload, variables || {}); const { couponCode, couponButtonIndex } = request.data || {}; if (couponCode) { payload.couponCode = couponCode; payload.couponButtonIndex = couponButtonIndex; } const res = await fetch(`${WIX_BASE}/sendTemplate`, { method: "POST", headers: { "Content-Type": "application/json", "X-Bridge-Secret": BRIDGE_SECRET.value() }, body: JSON.stringify({ data: payload }), }); const result = await res.json(); if (!res.ok || result.success === false) { throw new HttpsError("internal", "Template failed to send: " + JSON.stringify(result.results || result.error || result)); } const now = admin.firestore.Timestamp.now(); const outMsgId = `out_${Date.now()}_${Math.random().toString(36).slice(2)}`; const displayText = renderedText || `[Template: ${templateName}]`; await convRef.collection("messages").doc(outMsgId).set({ direction: "out", type: "template", text: displayText, status: "sent", source: "template", templateName, sentBy: agentEmail, createdAt: now, }); await convRef.set({ phone, customerName: customerName || conv?.customerName || phone, lastMessageText: displayText, lastMessageAt: now, lastMessageDirection: "out", unreadCount: 0, status: "open", updatedAt: now, }, { merge: true }); return { success: true }; } throw new HttpsError("invalid-argument", "mode must be 'text' or 'template'."); } ); // ───────────────────────────────────────────────────────────────────────── // getContacts — proxies Wix's new getContacts endpoint for the Customers tab. // ───────────────────────────────────────────────────────────────────────── exports.getContacts = onCall( { secrets: [BRIDGE_SECRET], region: "us-central1" }, async (request) => { if (!request.auth) throw new HttpsError("unauthenticated", "Sign in required."); const { search, limit } = request.data || {}; const res = await fetch(`${WIX_BASE}/getContacts`, { method: "POST", headers: { "Content-Type": "application/json", "X-Bridge-Secret": BRIDGE_SECRET.value() }, body: JSON.stringify({ search: search || "", limit: limit || 200 }), }); const result = await res.json(); if (!res.ok || result.success === false) { throw new HttpsError("internal", "Couldn't load contacts: " + JSON.stringify(result.error || result)); } return { contacts: result.contacts || [] }; } ); // ───────────────────────────────────────────────────────────────────────── // getTemplates — proxies Wix's new getTemplates endpoint (which calls Meta). // ───────────────────────────────────────────────────────────────────────── exports.getTemplates = onCall( { secrets: [BRIDGE_SECRET], region: "us-central1" }, async (request) => { if (!request.auth) throw new HttpsError("unauthenticated", "Sign in required."); const res = await fetch(`${WIX_BASE}/getTemplates`, { method: "POST", headers: { "Content-Type": "application/json", "X-Bridge-Secret": BRIDGE_SECRET.value() }, body: JSON.stringify({}), }); const result = await res.json(); if (!res.ok || result.success === false) { throw new HttpsError("internal", "Couldn't load templates: " + JSON.stringify(result.error || result)); } return { templates: result.templates || [] }; } ); // ───────────────────────────────────────────────────────────────────────── // editTemplate — proxies Wix's new editTemplate endpoint (which calls Meta). // Meta may reject edits to approved templates or require re-review — this // surfaces whatever Meta actually says rather than assuming success. // ───────────────────────────────────────────────────────────────────────── exports.editTemplate = onCall( { secrets: [BRIDGE_SECRET], region: "us-central1" }, async (request) => { if (!request.auth) throw new HttpsError("unauthenticated", "Sign in required."); const { templateId, components } = request.data || {}; if (!templateId || !components) throw new HttpsError("invalid-argument", "templateId and components are required."); const res = await fetch(`${WIX_BASE}/editTemplate`, { method: "POST", headers: { "Content-Type": "application/json", "X-Bridge-Secret": BRIDGE_SECRET.value() }, body: JSON.stringify({ templateId, components }), }); const result = await res.json(); if (!res.ok) throw new HttpsError("internal", "Couldn't reach the edit endpoint."); return result; // pass Meta's real response (success or rejection) straight through } ); // ───────────────────────────────────────────────────────────────────────── // sendBatchTemplate — sends a template to a hand-picked list of contacts, // personalizing {{customer_name}} per recipient. Calls Wix's EXISTING // sendTemplate endpoint once per contact — same code path as a single send, // just looped. Capped per call to avoid timeouts on very large lists; the // PLS HQ UI calls this in chunks and shows live progress. // ───────────────────────────────────────────────────────────────────────── exports.sendBatchTemplate = onCall( { secrets: [BRIDGE_SECRET], region: "us-central1", timeoutSeconds: 120 }, async (request) => { if (!request.auth) throw new HttpsError("unauthenticated", "Sign in required."); const { contacts, templateName, language, extraVars, couponCode, couponButtonIndex } = request.data || {}; if (!Array.isArray(contacts) || !contacts.length) throw new HttpsError("invalid-argument", "contacts must be a non-empty array."); if (contacts.length > 50) throw new HttpsError("invalid-argument", "Send at most 50 contacts per batch call."); if (!templateName) throw new HttpsError("invalid-argument", "templateName is required."); const results = []; for (const c of contacts) { const phone = cleanPhone(c.phone); if (!phone) { results.push({ phone: c.phone, success: false, error: "Invalid phone" }); continue; } const payload = { data: { to: phone, template: templateName, language: language || "en", customer_name: (c.customerName || "there").split(" ")[0], ...(extraVars || {}), ...(couponCode ? { couponCode, couponButtonIndex } : {}), }, }; try { const res = await fetch(`${WIX_BASE}/sendTemplate`, { method: "POST", headers: { "Content-Type": "application/json", "X-Bridge-Secret": BRIDGE_SECRET.value() }, body: JSON.stringify(payload), }); const result = await res.json(); const succeeded = res.ok && result.success !== false; results.push({ phone, success: succeeded, error: succeeded ? null : (result.results || result.error || result) }); } catch (e) { results.push({ phone, success: false, error: e.message }); } } return { results, sent: results.filter(r => r.success).length, failed: results.filter(r => !r.success).length }; } ); // ───────────────────────────────────────────────────────────────────────── // getMediaLink — resolves a fresh, working link for a voice note or // document, called on demand right when an agent opens one. Never stores // a link upfront for these, since Wix only gives out ones that expire // after ~10 hours. // ───────────────────────────────────────────────────────────────────────── exports.getMediaLink = onCall( { secrets: [BRIDGE_SECRET], region: "us-central1" }, async (request) => { if (!request.auth) throw new HttpsError("unauthenticated", "Sign in required."); const { fileUrl } = request.data || {}; if (!fileUrl) throw new HttpsError("invalid-argument", "fileUrl is required."); const res = await fetch(`${WIX_BASE}/getMediaLink`, { method: "POST", headers: { "Content-Type": "application/json", "X-Bridge-Secret": BRIDGE_SECRET.value() }, body: JSON.stringify({ fileUrl }), }); const result = await res.json(); if (!res.ok || result.success === false) { throw new HttpsError("internal", "Couldn't resolve a link: " + JSON.stringify(result.error || result)); } return { url: result.url }; } ); // ───────────────────────────────────────────────────────────────────────── // deleteConversation — removes a conversation from the PLS HQ Inbox, // including its messages and internal notes subcollections (which clients // can't delete directly by design, so this has to go through here). // // Deliberately does NOT touch Wix's IncomingMessages or WhatsAppLogs — // those stay as your permanent business record. This only clears the // PLS HQ / Firestore view of the conversation, not the underlying history. // ───────────────────────────────────────────────────────────────────────── async function deleteAllDocsInSubcollection(collRef) { let count = 0; while (true) { const snap = await collRef.limit(450).get(); if (snap.empty) break; const batch = db.batch(); snap.forEach((doc) => batch.delete(doc.ref)); await batch.commit(); count += snap.size; if (snap.size < 450) break; } return count; } exports.deleteConversation = onCall( { region: "us-central1" }, async (request) => { if (!request.auth) throw new HttpsError("unauthenticated", "Sign in required."); const { phone } = request.data || {}; if (!phone) throw new HttpsError("invalid-argument", "phone is required."); const convRef = db.collection("whatsapp_conversations").doc(phone); const deletedMessages = await deleteAllDocsInSubcollection(convRef.collection("messages")); const deletedNotes = await deleteAllDocsInSubcollection(convRef.collection("notes")); await convRef.delete(); return { success: true, deletedMessages, deletedNotes }; } );
